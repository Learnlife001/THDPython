# 1 add odd numbers up
def sum_odd_numbers(N):
    total = 0
    for number in range(1, N + 1):
        if number % 2 != 0:
            total += number
    return total
N = int(input("value for N: "))
result = sum_odd_numbers(N)
print("sum of odd numbers up to", N, "is:", result)


# 2 find the lowest number
def lowest_number(numbers):
    lowest = numbers[0]
    for number in numbers:
        if number < lowest:
            lowest = number
    return lowest
numbers_list = [1, 3, 5, 7, 9, 11]
lowest_number = lowest_number(numbers_list)
print("lowest number is:", lowest_number)


# 3 Fibinacci series
def fibonacci_series(n):
    fib_series = []
    a, b = 0, 1
    for _ in range(n):
        fib_series.append(a)
        a, b = b, a + b
    return fib_series
number_of_elements = 20
fib_sequence = fibonacci_series(number_of_elements)
print("The first", number_of_elements,
      "elements of the Fibonacci series are:", fib_sequence)


# 4 first ten Mersenne numbers
mersenne_numbers = []
for n in range(1, 11):
    mersenne_number = (2 ** n) - 1
    mersenne_numbers.append(mersenne_number)
print("first ten Mersenne numbers are:")
print(mersenne_numbers)

        print('*' * i)

# 6 leap year
def is_leap_year(year):
    if year % 4 == 0:
        if year % 100 == 0:
            if year % 400 == 0:
                return True
            else:
                return False
        else:
            return True
    else:
        return False
year_input = int(input("Enter a year: "))
if is_leap_year(year_input):
    print(f"{year_input} is a Leap year.")
else:
    print(f"{year_input} is not a Leap year.")


# 7 three sides of a triangle
def can_form_triangle(a, b, c):
    return (a + b > c) and (b + c > a) and (c + a > b)
side1 = float(input("side 1: "))
side2 = float(input("side 2: "))
side3 = float(input("side 3: "))
if can_form_triangle(side1, side2, side3):
    print("The sides can form a triangle.")
else:
    print("The sides cannot form a triangle.")


# 8 digit to word
def digit_to_word(digit):
    digit_words = {'0': 'zero', '1': 'one', '2': 'two', '3': 'three', '4': 'four',
                   '5': 'five', '6': 'six', '7': 'seven', '8': 'eight', '9': 'nine'}
    return digit_words.get(digit, "Invalid input. Please enter a single digit (0-9).")
user_input = input("Enter a single digit (0-9): ")
print(digit_to_word(user_input))


# 9 palindrome checker
def is_palindrome(input_str):
    cleaned_str = ''.join(input_str.split()).lower()
    return cleaned_str == cleaned_str[::-1]
user_input = input("Enter a string or number to check if it's a palindrome: ")
if is_palindrome(user_input):
    print(f'"{user_input}" is a palindrome.')
else:
    print(f'"{user_input}" is not a palindrome.')


# 10 temperature converter
def convert_temperature(temp, unit):
    if unit.upper() == 'C':
        converted_temp = (9/5) * temp + 32
        return converted_temp, 'Fahrenheit'
    elif unit.upper() == 'F':
        converted_temp = (5/9) * (temp - 32)
        return converted_temp, 'Celsius'
    else:
        return None, None
temperature = float(input("Enter the temperature: "))  # Convert input to float
unit = input("Is this temperature in Celsius or Fahrenheit? (C/F): ")
converted_temp, converted_unit = convert_temperature(temperature, unit)
if converted_temp is not None:
    print(f"{temperature}° {unit.upper()} is {converted_temp:.2f}° {converted_unit}.")
else:
    print("Invalid unit entered. Please use 'C' for Celsius or 'F' for Fahrenheit.")
